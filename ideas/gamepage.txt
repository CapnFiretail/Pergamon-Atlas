    const pageName = "Games";
    const pageHeaderSuffix = pageName ? " | " + pageName : "";

      .then(res => res.text())
      .then(data => {
        document.getElementById('header-placeholder').innerHTML = data;

        const pageTitleText = document.querySelector('.page-title-text');
        if (pageTitleText) {
          pageTitleText.textContent = pageHeaderSuffix;
        }
      });

    fetch('/snippets/sidebar.html')
      .then(res => res.text())
      .then(data => {
        document.getElementById('sidebar-placeholder').innerHTML = data;

        const path = window.location.pathname;

        document.querySelectorAll('.sidebar a').forEach(link => {
          link.classList.remove('active');

          if (path === "/" && link.getAttribute("href") === "/") {
            link.classList.add('active');
          }
          else if (path.startsWith("/tools") && link.getAttribute("href") === "/tools") {
            link.classList.add('active');
          }
          else if (path.startsWith("/games") && link.getAttribute("href") === "/games") {
            link.classList.add('active');
          }
          else if (
            (path.startsWith("/help") || path.startsWith("/pages/help")) &&
            (link.getAttribute("href") === "/help" || link.getAttribute("href") === "/pages/help")
          ) {
            link.classList.add('active');
          }
        });

        (adsbygoogle = window.adsbygoogle || []).push({});
      });

    fetch('/snippets/footer.html')
      .then(res => res.text())
      .then(data => {
        document.getElementById('footer-placeholder').innerHTML = data;
      });
